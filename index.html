async function processImage() { const fileInput = document.getElementById("fileInput"); const file = fileInput.files[0]; if (!file) { alert("Pilih foto dulu!"); return; } const formData = new FormData(); formData.append("image", file); // Tambah timeout const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 30000); // 30 detik try { setLoading(true); const response = await fetch("/api/remove-bg", { method: "POST", body: formData, signal: controller.signal }); clearTimeout(timeoutId); if (!response.ok) { const errorData = await response.json().catch(() => ({})); throw new Error(errorData.error || `Error ${response.status}`); } const blob = await response.blob(); const url = URL.createObjectURL(blob); result.src = url; result.style.display = "block"; downloadBtn.href = url; downloadBtn.style.display = "inline-block"; } catch (error) { if (error.name === 'AbortError') { alert("Proses timeout! Server terlalu lama merespons. Coba lagi."); } else { alert("Error: " + error.message); } console.error("Error detail:", error); } finally { setLoading(false); } }
